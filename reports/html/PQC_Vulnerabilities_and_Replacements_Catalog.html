🔑 1. ML-KEM-768 Key Encapsulation (FIPS 203)
Replaces: RSA Key Exchange, Diffie-Hellman (DH), ECDH 
import oqs  # liboqs-python (Open Quantum Safe)

def pqc_key_exchange():
    kem_algo = "ML-KEM-768"  # NIST Security Level 3
    
    # Client: Generate public/private keypair
    with oqs.KeyEncapsulation(kem_algo) as client:
        client_pubkey = client.generate_keypair()
        
        # Server: Encapsulate shared secret against client public key
        with oqs.KeyEncapsulation(kem_algo) as server:
            ciphertext, secret_server = server.encap_secret(client_pubkey)
            
        # Client: Decapsulate ciphertext to recover identical secret
        secret_client = client.decap_secret(ciphertext)
        
    assert secret_client == secret_server
    return secret_client, client_pubkey, ciphertext
✍️ 2. ML-DSA-65 Digital Signatures (FIPS 204)
Replaces: RSA Signatures, ECDSA, DSA, Ed25519 
import oqs  # liboqs-python (Dilithium / ML-DSA)

def pqc_sign(message: bytes):
    sig_algo = "ML-DSA-65"  # NIST Security Level 3
    with oqs.Signature(sig_algo) as signer:
        pubkey = signer.generate_keypair()
        signature = signer.sign(message)
    return signature, pubkey

def pqc_verify(message: bytes, sig: bytes, pubkey: bytes) -> bool:
    with oqs.Signature("ML-DSA-65") as verifier:
        return verifier.verify(message, sig, pubkey)
🛡️ 3. AES-256-GCM Symmetric Hardening

import os
from cryptography.hazmat.primitives.ciphers.aead import AESGCM

def encrypt_aes256_gcm(plaintext: bytes, aad: bytes = None):
    # 256-bit key maintains 128-bit security under Grover's algorithm
    key = AESGCM.generate_key(bit_length=256)
    nonce = os.urandom(12)  # 96-bit unique IV/nonce
    aesgcm = AESGCM(key)
    ciphertext = aesgcm.encrypt(nonce, plaintext, aad)
    return ciphertext, nonce, key

def decrypt_aes256_gcm(ciphertext: bytes, nonce: bytes, key: bytes):
    return AESGCM(key).decrypt(nonce, ciphertext, None)
